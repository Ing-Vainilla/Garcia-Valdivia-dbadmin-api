enum estado_cliente {
  ACTIVO
  INACTIVO
}

enum tipo_direccion {
  ORIGEN
  DESTINO
}

enum estado_envio {
  REGISTRADO
  EN_TRANSITO
  ENTREGADO
  CANCELADO
}

Table clientes {
  id int [pk, increment]
  nombre varchar(100) [not null]
  email varchar(100) [unique, not null]
  telefono varchar(20)
  estado estado_cliente [default: 'ACTIVO', not null]
  limite_credito decimal(10,2) [default: 0.00, not null]
  created_at datetime [default: `now()`]
}

Table direcciones {
  id int [pk, increment]
  cliente_id int [not null]
  calle varchar(150) [not null]
  numero varchar(20) [not null]
  ciudad varchar(100) [not null]
  estado varchar(100) [not null]
  codigo_postal varchar(10) [not null]
  tipo tipo_direccion [not null]
  created_at datetime [default: `now()`]
}

Table almacenes {
  id int [pk, increment]
  nombre varchar(100) [not null]
  ubicacion varchar(255) [not null]
  capacidad_maxima int [not null]
}

Table envios {
  id int [pk, increment]
  cliente_id int [not null]
  destinatario varchar(100) [not null]
  direccion_origen_id int [not null]
  direccion_destino_id int [not null]
  costo_total decimal(10,2) [not null]
  fecha datetime [default: `now()`, not null]
  fecha_entrega datetime [note: 'Registrada al momento de entrega para cálculo de tiempos']
  estado estado_envio [default: 'REGISTRADO', not null]
}

Table paquetes {
  id int [pk, increment]
  envio_id int [not null]
  almacen_id int [null]
  peso decimal(8,2) [not null]
  dimensiones varchar(50) [not null]
  contenido varchar(255) [not null]
}

Table historial_rastreo {
  id int [pk, increment]
  envio_id int [not null]
  fecha datetime [default: `now()`, not null]
  ubicacion_actual varchar(255) [not null]
  estado_evento varchar(50) [not null]
  observaciones text
}

Table auditoria_envios {
  id int [pk, increment]
  envio_id int [not null]
  costo_anterior decimal(10,2) [not null]
  costo_nuevo decimal(10,2) [not null]
  usuario varchar(100) [not null]
  fecha_modificacion datetime [default: `now()`, not null]
}
